# Команды Git для создания репозитория на GitHub
# Скопируйте и выполните команды последовательно

# 1. Инициализация репозитория
git init

# 2. Добавление всех файлов
git add .

# 3. Первый коммит
git commit -m "Initial commit: Проект по эконометрике - исследование гендерного разрыва в заработных платах на данных RLMS"

# 4. Переименование ветки в main (если нужно)
git branch -M main

# 5. Добавление remote репозитория
git remote add origin https://github.com/VenistorAlishi/gender-wage-gap-rlms.git

# 6. Отправка кода на GitHub
git push -u origin main

# Дополнительные команды:

# Проверка статуса
git status

# Просмотр изменений
git diff

# Просмотр истории коммитов
git log --oneline
